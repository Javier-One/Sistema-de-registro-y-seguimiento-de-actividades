function handleReq(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { if (e.parameter && e.parameter.action === 'getActivities') { return respond(e, getActivities()); } let data; if (e.parameter && e.parameter.data) { data = JSON.parse(e.parameter.data); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else { return respond(e, {status:'activo'}); } if (data.action === 'addActivity') return respond(e, addActivity(data.data)); if (data.action === 'deleteActivity') return respond(e, deleteActivity(data.data)); if (data.action === 'editActivity') return respond(e, editActivity(data.data)); if (data.action === 'getActivities') return respond(e, getActivities()); return respond(e, {status:'error', mensaje:'Accion no reconocida'}); } catch(err) { return respond(e, {status:'error', mensaje: err.toString()}); } finally { lock.releaseLock(); } } function deleteActivity(d) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); const sheetsRow = parseInt(d.sheetsRow); if (sheetsRow >= 3 && sheetsRow <= sheet.getLastRow()) { sheet.deleteRow(sheetsRow); } return {status: 'ok'}; } function editActivity(d) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); const sheetsRow = parseInt(d.sheetsRow); if (sheetsRow >= 3 && sheetsRow <= sheet.getLastRow()) { sheet.getRange(sheetsRow, 2, 1, 10).setValues([[ d.fecha, d.persona, d.cargo, d.actividad, d.tipo, d.horaInicio, d.horaFin, d.horas, d.tipo, d.observaciones || '' ]]); } return {status: 'ok'}; }
