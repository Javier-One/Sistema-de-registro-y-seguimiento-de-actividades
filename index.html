// ══════════════════════════════════════════════════════════════ // APPS SCRIPT — Sistema TI PagoDirecto v4 // Columnas del Sheets (fila 3 en adelante): // A=ID B=Fecha C=Persona D=Cargo E=Actividad // F=Tipo G=Horas H=Observaciones I=Proyecto/Plataforma // ══════════════════════════════════════════════════════════════ const SHEET_NAME = 'Actividades'; const CONFIG_NAME = 'Config'; const DASH_NAME = 'Dashboard'; const SPREADSHEET_ID = 'PEGAR_AQUI_EL_ID_DE_TU_SHEETS'; function doGet(e) { return handleReq(e); } function doPost(e) { return handleReq(e); } function handleReq(e) { const lock = LockService.getScriptLock(); lock.tryLock(15000); try { const param = e.parameter && e.parameter.action; if (param === 'getActivities') return respond(e, getActivities()); if (param === 'getConfig') return respond(e, getConfig()); let data; if (e.parameter && e.parameter.data) { data = JSON.parse(e.parameter.data); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else { return respond(e, {status:'activo', mensaje:'Apps Script v3 activo'}); } if (data.action === 'getConfig') return respond(e, getConfig()); if (data.action === 'saveConfig') return respond(e, saveConfig(data.data)); if (data.action === 'addActivity') return respond(e, addActivity(data.data)); if (data.action === 'editActivity') return respond(e, editActivity(data.data)); if (data.action === 'deleteActivity') return respond(e, deleteActivity(data.data)); if (data.action === 'getActivities') return respond(e, getActivities()); return respond(e, {status:'error', mensaje:'Accion no reconocida: '+data.action}); } catch(err) { return respond(e, {status:'error', mensaje: err.toString()}); } finally { lock.releaseLock(); } } function respond(e, data) { const cb = e.parameter && e.parameter.callback; const output = cb ? cb+'('+JSON.stringify(data)+');' : JSON.stringify(data); const mime = cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON; return ContentService.createTextOutput(output).setMimeType(mime); } function getSheet() { return SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME); // Centrar la fila recién agregada try { const nr = sheet.getLastRow(); sheet.getRange(nr, 1, 1, 9).setHorizontalAlignment('center'); } catch(e) {} try { rebuildDashboard(); } catch(e) { Logger.log('Dashboard error: '+e); } return {status:'ok', id: newId}; } // ── EDITAR ACTIVIDAD ────────────────────────────────────────── function editActivity(d) { const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 3) return {status:'error', mensaje:'Sin datos'}; const buscarId = String(d.actId !== undefined ? d.actId : d.id); const ids = sheet.getRange(3, 1, lastRow-2, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]) === buscarId) { sheet.getRange(i+3, 1, 1, 9).setValues([[ buscarId, d.fecha, d.persona, d.cargo, d.actividad, d.tipo, d.horas, d.observaciones||'', d.proyecto||'' ]]); return h > 0 ? WHITE : GRIS) .setFontSize(10); // Negrita nombre, centrar números dash.getRange(row, 1).setFontWeight('bold'); dash.getRange(row, 3).setFontColor(VERDE_OSCURO).setFontWeight('bold').setHorizontalAlignment('center'); dash.getRange(row, 4).setHorizontalAlignment('center'); if(tipoVals.length>0) dash.getRange(row, 5, 1, tipoVals.length).setHorizontalAlignment('center'); row++; }); // Fila TOTAL if (personas.length > 0) { const totalHoras = personas.reduce((s,[,d])=>s+d.horas,0); const totalActs = personas.reduce((s,[,d])=>s+d.acts,0); const totalTipos = tipos.map(t => personas.reduce((s,[,d])=>s+(d.porTipo[t]||0),0)); const totalRow = ['TOTAL EQUIPO', '', Math.round(totalHoras), totalActs] .concat(totalTipos.map(v=>Math.round(v))); dash.getRange(row, 1, 1, totalRow.length) .setValues([totalRow]) .setBackground(VERDE_CLARO) .setFontWeight('bold') .setFontColor(VERDE_OSCURO) .setFontSize(10); row++; } row += 2; // ── SECCIÓN 2: TOTALES POR TIPO ── dash.getRange(row, 1, 1, 4).merge() .setValue('TOTALES POR TIPO DE ACTIVIDAD') .setBackground(AZUL_CLARO.replace('fb','90')) .setFontColor('#1a3a6a') .setFontWeight('bold') .setFontSize(11); row++; const tipoHeaders2 = ['Tipo de Actividad', 'Total Horas', '% del Total', '# Actividades']; dash.getRange(row, 1, 1, 4) .setValues([tipoHeaders2]) .setBackground('#2a5fa8') .setFontColor(WHITE) .setFontWeight('bold') .setFontSize(9) .setHorizontalAlignment('center'); row++; const totalHorasAll = rows.reduce((s,r)=>s+(parseFloat(r[6])||0),0); tipos.forEach((tipo, idx) => { const horasTipo = rows.filter(r=>String(r[5])===tipo).reduce((s,r)=>s+(parseFloat(r[6])||0),0); const actsTipo = rows.filter(r=>String(r[5])===tipo).length; const pct = totalHorasAll > 0 ?
